from django.contrib.auth.models import User
from sales_app.models import UserProfile

# Get the admin user

admin.set_password('admin123')
admin.save()

print("✓ Password reset for admin")

# Make sure profile exists
profile, created = UserProfile.objects.get_or_create(
    user=admin,
    defaults={'is_admin': True}
)
if created:
    print("✓ Profile created")
else:
    print("✓ Profile already exists")

import pandas as pd
import psycopg2
from psycopg2.extras import execute_values
from datetime import date

# --- CONFIG ---
PKL_PATH = r'/Users/macintoshi/Downloads/SBS_MAIN.pkl'
DB_CONFIG = {
    'host': 'localhost',
    'database': 'main_db',
    'user': 'postgres',       # fixed typo
    'password': 'overall',
    'port': 5432
}

START_DATE = '2024-01-01'
END_DATE = date.today().strftime('%Y-%m-%d')  # today dynamically

COLUMNS = [
    'IdReal1', 'IdReal2', 'Zedd', 'CD', 'UN', 'IdTanam', 'IdProd',
    'IdActions', 'raod', 'discount_price', 'Sachuqari', 'std_price',
    'Tanxa', 'Prod', 'IdProdT', 'IdProdG', 'Desc1', 'ProdT', 'ProdG',
    'Actions', 'Segment', 'IdGvari', 'Gvari', 'Tanam'
]

BATCH_SIZE = 1000

# --- LOAD PKL ---
print(f"Loading PKL from {PKL_PATH}...")
df = pd.read_pickle(PKL_PATH)
print(f"Loaded {len(df)} rows")

# --- FILTER DATE RANGE ---
df['CD'] = pd.to_datetime(df['CD'])
df = df[(df['CD'] >= START_DATE) & (df['CD'] <= END_DATE)]
print(f"Filtered to {len(df)} rows ({START_DATE} to {END_DATE})")

# --- SELECT COLUMNS ---
df = df[COLUMNS]

# --- CONNECT ---
print("Connecting to database...")
conn = psycopg2.connect(**DB_CONFIG)
cur = conn.cursor()

# --- DELETE EXISTING RANGE ---
cur.execute("""
    DELETE FROM sales_main_web 
    WHERE "CD" >= %s AND "CD" <= %s
""", (START_DATE, END_DATE))
print(f"Deleted {cur.rowcount} existing rows in range")

# --- BATCH INSERT ---
col_str = ', '.join(f'"{c}"' for c in COLUMNS)
total_inserted = 0

for i in range(0, len(df), BATCH_SIZE):
    batch = df.iloc[i:i + BATCH_SIZE]
    rows = [tuple(row) for row in batch.itertuples(index=False)]
    execute_values(cur, f"""
        INSERT INTO sales_main_web ({col_str})
        VALUES %s
    """, rows)
    total_inserted += len(rows)
    print(f"  Inserted {total_inserted}/{len(df)} rows...", end='\r')

conn.commit()
cur.close()
conn.close()

print(f"\nDone! Inserted {total_inserted} rows from {START_DATE} to {END_DATE}")
